** 起動の高速化

[[https://www.takeokunn.org/posts/permanent/20221217132815-emacs_more_faster/][Emacsを世界最速級で起動する方法]] に倣い、起動中だけGCとファイル名ハンドラーの照合を止める。
しきい値を大きいままにすると、起動後のGCが一度に多くを回収して止まる時間が長くなるため、起動後に元へ戻す。
起動中にパッケージが足したハンドラーを失わないよう、ハンドラーは上書きせずにまぜて戻す。

#+begin_src emacs-lisp :tangle early-init.el
(let ((saved-gc-cons-threshold gc-cons-threshold)
      (saved-file-name-handler-alist file-name-handler-alist))
  (setq gc-cons-threshold (* 128 1024 1024)
        file-name-handler-alist nil)
  (add-hook 'emacs-startup-hook
            (lambda ()
              (setq gc-cons-threshold saved-gc-cons-threshold
                    file-name-handler-alist
                    (delete-dups (append file-name-handler-alist
                                         saved-file-name-handler-alist))))))
#+end_src

** 基本のキーマップ

=vimrc.core= のキーマップに合わせる。
=cw= が単語の末尾で止まるように、レジスタを汚さない =c= を =evil-change-commands= に加える。

#+begin_src emacs-lisp :tangle init.el
(defun +vimrc--only-indent-before-point-p ()
  (string-match-p "\\`[ \t]+\\'"
                  (buffer-substring-no-properties (line-beginning-position) (point))))

(defun +vimrc--insert-with-indent ()
  (evil-insert-state)
  (indent-according-to-mode))

(with-eval-after-load 'evil
  (evil-define-motion +vimrc/next-line (count)
    :type line
    (if count (evil-next-line count) (evil-next-visual-line)))

  (evil-define-motion +vimrc/previous-line (count)
    :type line
    (if count (evil-previous-line count) (evil-previous-visual-line)))

  (evil-define-motion +vimrc/beginning-of-line-or-indentation ()
    :type exclusive
    (if (+vimrc--only-indent-before-point-p)
        (evil-beginning-of-line)
      (evil-first-non-blank)))

  (evil-define-operator +vimrc/delete-char-without-yank (beg end type)
    :motion evil-forward-char
    (interactive "<R>")
    (evil-delete beg end type ?_))

  (evil-define-operator +vimrc/substitute-without-yank (beg end type)
    :motion evil-forward-char
    (interactive "<R>")
    (evil-change beg end type ?_))

  (evil-define-operator +vimrc/change-without-yank (beg end type)
    (interactive "<R>")
    (evil-change beg end type ?_))
  (add-to-list 'evil-change-commands #'+vimrc/change-without-yank)

  (evil-define-command +vimrc/insert (count)
    (interactive "p")
    (if (= (line-beginning-position) (line-end-position))
        (+vimrc--insert-with-indent)
      (evil-insert count)))

  (evil-define-command +vimrc/append-line (count)
    (interactive "p")
    (if (= (line-beginning-position) (line-end-position))
        (+vimrc--insert-with-indent)
      (evil-append-line count))))

(defun +vimrc/insert-beginning-of-line-or-indentation ()
  (interactive)
  (if (+vimrc--only-indent-before-point-p)
      (beginning-of-line)
    (back-to-indentation)))

(defun +vimrc/substitute-in-buffer ()
  (interactive)
  (+vimrc--ex-substitute "%"))

(defun +vimrc/substitute-in-region ()
  (interactive)
  (+vimrc--ex-substitute "'<,'>"))

(defun +vimrc--ex-substitute (range)
  (minibuffer-with-setup-hook (lambda () (backward-char 3))
    (evil-ex (concat range "s@\\v@@g"))))

(defun +vimrc/close-folds-except-cursor ()
  (interactive)
  (evil-close-folds)
  (cl-loop repeat 100
           while (invisible-p (point))
           do (evil-open-fold)))

(with-eval-after-load 'evil
  (evil-define-key* '(normal visual) 'global
    "j"  #'+vimrc/next-line
    "k"  #'+vimrc/previous-line
    "gj" #'evil-join
    "J"  #'evil-next-visual-line)
  (evil-define-key* 'visual 'global
    "K"          #'evil-previous-visual-line
    (kbd "C-s")  #'+vimrc/substitute-in-region)
  (evil-define-key* 'normal 'global
    "x"          #'+vimrc/delete-char-without-yank
    "s"          #'+vimrc/substitute-without-yank
    "c"          #'+vimrc/change-without-yank
    "0"          #'+vimrc/beginning-of-line-or-indentation
    "i"          #'+vimrc/insert
    "A"          #'+vimrc/append-line
    (kbd "C-s")  #'+vimrc/substitute-in-buffer
    "z="         #'+vimrc/close-folds-except-cursor
    "[b"         #'next-buffer
    "]b"         #'previous-buffer)
  (evil-define-key* 'operator 'global
    "A" #'evil-end-of-line)
  (evil-define-key* 'insert 'global
    (kbd "C-f") #'forward-char
    (kbd "C-b") #'backward-char
    (kbd "C-a") #'+vimrc/insert-beginning-of-line-or-indentation
    (kbd "C-e") #'end-of-line)
  (evil-define-key* '(normal visual operator insert) 'global
    [left]  #'ignore
    [right] #'ignore
    [up]    #'ignore
    [down]  #'ignore))
#+end_src

** 全角文字へのf

Neovimの =fj= と同じく、 =f= のあとに =j= を打つと、もう1文字読んでdigraphで全角文字に変換してから探す。
Emacsでは =fj= をキーに割り当てると =f= 単体が使えなくなるため、 =f= 自体を置きかえる。

#+begin_src emacs-lisp :tangle init.el
(setq evil-digraphs-table-user
      '(((?j ?j) . ?j)
        ((?j ?\() . ?（)
        ((?j ?\)) . ?）)
        ((?j ?\[) . ?「)
        ((?j ?\]) . ?」)
        ((?j ?,) . ?、)
        ((?j ?.) . ?。)
        ((?j ?!) . ?！)
        ((?j ??) . ?？)
        ((?j ?~) . ?～)))

(defun +vimrc--read-find-char ()
  (let ((char (evil-read-key)))
    (if (eq char ?j)
        (let ((next (evil-read-key)))
          (require 'evil-digraphs)
          (or (evil-digraph (list ?j next)) next))
      char)))

(with-eval-after-load 'evil
  (evil-define-motion +vimrc/find-char (count char)
    :type inclusive
    (interactive "<c>" (list (+vimrc--read-find-char)))
    (evil-find-char count char))
  (evil-define-key* '(normal visual operator) 'global
    "f" #'+vimrc/find-char))
#+end_src

** エラー一覧の移動

Neovimの =qn= / =qN= はquickfixを移動し、続けて =n= / =N= を押すとさらに移動する。
Emacsでは =next-error= で同じことをする。
=q= をプレフィックスにするとマクロの記録ができなくなるため、 =q= のあとのキーが =n= / =N= のときだけ移動し、それ以外はマクロの記録に回す。
=Q= はquickfixを開く代わりに、 =next-error= の対象バッファを表示する。

#+begin_src emacs-lisp :tangle init.el
(defvar-keymap +vimrc-next-error-repeat-map
  "n" #'next-error
  "N" #'previous-error)

(defun +vimrc/display-next-error-buffer ()
  (interactive)
  (display-buffer (or (next-error-find-buffer)
                      (user-error "No buffer to navigate errors in"))))

(with-eval-after-load 'evil
  (evil-define-command +vimrc/record-macro-or-navigate-errors (register)
    :keep-visual t
    :suppress-operator t
    (interactive
     (list (unless (and evil-this-macro defining-kbd-macro)
             (or evil-this-register (evil-read-key)))))
    (pcase register
      (?n (next-error)
          (set-transient-map +vimrc-next-error-repeat-map t))
      (?N (previous-error)
          (set-transient-map +vimrc-next-error-repeat-map t))
      (_ (evil-record-macro register))))
  (evil-define-key* 'normal 'global
    "q" #'+vimrc/record-macro-or-navigate-errors
    "Q" #'+vimrc/display-next-error-buffer))
#+end_src

** Exコマンド

Neovimと同じく、 =:w= は変更があるときだけ保存し、 =:ww= で強制的に保存、 =:qq= で強制的に閉じる。

=:h= は、Neovimと同じく引数の名前のヘルプを開き、引数がなければヘルプの一覧を出す。
evilは =:help= を持たず、同じ名前のEmacsの =help= を呼ぶため、引数を渡せない。
関数と変数のどちらの名前でも開けるよう、 =describe-symbol= を使う。

#+begin_src emacs-lisp :tangle init.el
(defun +vimrc/force-quit ()
  (interactive)
  (evil-quit t))

(with-eval-after-load 'evil
  (evil-define-command +vimrc/help (topic)
    (interactive "<a>")
    (if topic
        (describe-symbol (or (intern-soft topic)
                             (user-error "No help for %s" topic)))
      (help-for-help)))
  (evil-ex-define-cmd "w[rite]" #'evil-update)
  (evil-ex-define-cmd "ww" #'evil-write)
  (evil-ex-define-cmd "qq" #'+vimrc/force-quit)
  (evil-ex-define-cmd "h[elp]" #'+vimrc/help))
#+end_src

* nskk

日本語入力にはnskkを用い、AZIK配列を前提とした設定を行う。
ddskkはneomacsの =emacsPackagesFor= でパッケージを組み立てる途中で文字コードを問い合わせて止まるため、使わない。

** 基本設定

大辞書には、NixでUTF-8に変換したSKK-JISYO.Lを使う。
=@skkJisyoL@= は、tangleするときにNixがそのパスへ置き換える。
有効にした直後からかな入力できるよう、初期モードはひらがなにする。

#+begin_src emacs-lisp :tangle init.el
(use-package nskk
  :commands nskk-mode
  :config
  (setopt nskk-dict-system-dictionary-files '("@skkJisyoL@")
          nskk-converter-romaji-style 'azik
          nskk-azik-keyboard-type 'jp106
          nskk-state-default-mode 'hiragana))
#+end_src

挿入モードの =C-j= は、nskkが無効なら有効にし、有効なら =nskk-kakutei= で確定する。
evilの挿入モードのキーマップはnskkのキーマップより優先されるため、確定もこのコマンドで受け持つ。

#+begin_src emacs-lisp :tangle init.el
(defun my/nskk-enable-or-kakutei ()
  "Enable nskk in the current buffer, or commit when it is already enabled."
  (interactive)
  (if (bound-and-true-p nskk-mode)
      (nskk-kakutei)
    (nskk-mode 1)))

(with-eval-after-load 'evil
  (evil-define-key* 'insert 'global (kbd "C-j") #'my/nskk-enable-or-kakutei))
#+end_src

** 辞書の先読み

nskkは最初に有効にしたときに辞書をまとめて読むため、最初の =C-j= に2秒ほどかかる。
起動を待たせないよう、起動後に手が止まったところで、使い捨てのバッファで一度有効にして辞書を読んでおく。
辞書を読んだあとは、有効にするのにかかる時間が40msほどになる。
アイドルタイマーは起動が終わるまで動かないため、 =emacs-startup-hook= で包まなくてよい。

#+begin_src emacs-lisp :tangle init.el
(run-with-idle-timer 1 nil
                     (lambda ()
                       (with-temp-buffer
                         (nskk-mode 1))))
#+end_src

** AZIKのかな入力規則カスタマイズ

=nskk-converter-user-rules= はAZIKの規則より優先されるため、ここに足した規則で既定の規則を上書きできる。
=xxa= から =xxo= の小書きのかなはnskkのAZIKに最初からあるため、足りない拗音と「ゎ」だけを足す。
=nn= は、skkeletonの設定に揃えて「なん」にする。

#+begin_src emacs-lisp :tangle init.el
(with-eval-after-load 'nskk
  (setopt nskk-converter-user-rules '(("xxya" . "ゃ")
                                      ("xxyu" . "ゅ")
                                      ("xxyo" . "ょ")
                                      ("xxwa" . "ゎ")
                                      ("rr" . "れる")
                                      ("nn" . "なん"))))
#+end_src

ただしnskkは、 =n= に続く =n= を規則より先に =nn-double= と分類して「ん」を出すため、規則だけでは =nn= が効かない。
分類する内部関数にadviceを足し、 =nn= を規則の結果で変換させる。
内部関数に頼るため、nskkの更新で関数名や分類の名前が変わると効かなくなる。

#+begin_src emacs-lisp :tangle init.el
(defun my/nskk-prefer-user-nn-rule (classify char last-buf-char result)
  "Let the user rule for nn win over the built-in nn handling of nskk."
  (let ((class (funcall classify char last-buf-char result)))
    (if (eq class 'nn-double)
        'match
      class)))

(with-eval-after-load 'nskk
  (advice-add 'nskk--classify-romaji-input :around #'my/nskk-prefer-user-nn-rule))
#+end_src

=:= はAZIK既定の「ー」をやめ、skkeletonのhenkanPointと同じsticky shiftに割り当てる。
何もないときに押すと読みを始め、読みの途中で押すと送りがなを区切る。
nskkの規則はかなの文字列しか返せないため、規則ではなくキーマップで割り当てる。
sticky shiftは =nskk-handle-semicolon-key= にあるが、AZIKでは「っ」の入力になるため、標準のローマ字のつもりで呼び出す。
このコマンドは押した時点のローマ字の方式を見て動きを決めるため、それに頼っている。
=Q= と同じ処理にすると、読みの途中で押したときに入力が崩れる。

#+begin_src emacs-lisp :tangle init.el
(defun my/nskk-set-henkan-point ()
  "Act as sticky shift regardless of the AZIK style."
  (interactive)
  (let ((nskk-converter-romaji-style 'standard))
    (nskk-handle-semicolon-key)))

(with-eval-after-load 'nskk
  (keymap-set nskk-mode-map ":" #'my/nskk-set-henkan-point))
#+end_src

   '(:application tramp :protocol "ssh")
   'remote-direct-async-process))
#+end_src

* 端末

=emacs -nw= で使うための設定を置く。
daemonはGUIと端末のフレームを同時に持つため、起動時に =display-graphic-p= で分けず、端末だけで効くパッケージに任せる。

クリップボードはNeovimと同じく、手元では =pbcopy= などのコマンドを使い、SSHの先ではOSC 52で書き込む。

#+begin_src emacs-lisp :tangle init.el
(use-package evil-terminal-cursor-changer
  :after evil
  :config
  (evil-terminal-cursor-changer-activate))

(use-package kkp
  :config
  (global-kkp-mode 1))

(use-package xclip
  :unless (getenv "SSH_TTY")
  :config
  (xclip-mode 1))

(use-package clipetty
  :if (getenv "SSH_TTY")
  :config
  (global-clipetty-mode 1))
