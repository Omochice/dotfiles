        evil-search-module 'evil-search

        evil-symbol-word-search t

  :config

** 基本のキーマップ

=vimrc.core= のキーマップに合わせる。
=cw= が単語の末尾で止まるように、レジスタを汚さない =c= を =evil-change-commands= に加える。

#+begin_src emacs-lisp :tangle init.el
(defun +vimrc--only-indent-before-point-p ()
  (string-match-p "\\`[ \t]+\\'"
                  (buffer-substring-no-properties (line-beginning-position) (point))))

(defun +vimrc--insert-with-indent ()
  (evil-insert-state)
  (indent-according-to-mode))

(with-eval-after-load 'evil
  (evil-define-motion +vimrc/next-line (count)
    :type line
    (if count (evil-next-line count) (evil-next-visual-line)))

  (evil-define-motion +vimrc/previous-line (count)
    :type line
    (if count (evil-previous-line count) (evil-previous-visual-line)))

  (evil-define-motion +vimrc/beginning-of-line-or-indentation ()
    :type exclusive
    (if (+vimrc--only-indent-before-point-p)
        (evil-beginning-of-line)
      (evil-first-non-blank)))

  (evil-define-operator +vimrc/delete-char-without-yank (beg end type)
    :motion evil-forward-char
    (interactive "<R>")
    (evil-delete beg end type ?_))

  (evil-define-operator +vimrc/substitute-without-yank (beg end type)
    :motion evil-forward-char
    (interactive "<R>")
    (evil-change beg end type ?_))

  (evil-define-operator +vimrc/change-without-yank (beg end type)
    (interactive "<R>")
    (evil-change beg end type ?_))
  (add-to-list 'evil-change-commands #'+vimrc/change-without-yank)

  (evil-define-command +vimrc/insert (count)
    (interactive "p")
    (if (= (line-beginning-position) (line-end-position))
        (+vimrc--insert-with-indent)
      (evil-insert count)))

  (evil-define-command +vimrc/append-line (count)
    (interactive "p")
    (if (= (line-beginning-position) (line-end-position))
        (+vimrc--insert-with-indent)
      (evil-append-line count))))

(defun +vimrc/insert-beginning-of-line-or-indentation ()
  (interactive)
  (if (+vimrc--only-indent-before-point-p)
      (beginning-of-line)
    (back-to-indentation)))

(defun +vimrc/substitute-in-buffer ()
  (interactive)
  (+vimrc--ex-substitute "%"))

(defun +vimrc/substitute-in-region ()
  (interactive)
  (+vimrc--ex-substitute "'<,'>"))

(defun +vimrc--ex-substitute (range)
  (minibuffer-with-setup-hook (lambda () (backward-char 3))
    (evil-ex (concat range "s@\\v@@g"))))

(defun +vimrc/close-folds-except-cursor ()
  (interactive)
  (evil-close-folds)
  (cl-loop repeat 100
           while (invisible-p (point))
           do (evil-open-fold)))

(with-eval-after-load 'evil
  (evil-define-key* '(normal visual) 'global
    "j"  #'+vimrc/next-line
    "k"  #'+vimrc/previous-line
    "gj" #'evil-join
    "J"  #'evil-next-visual-line)
  (evil-define-key* 'visual 'global
    "K"          #'evil-previous-visual-line
    (kbd "C-s")  #'+vimrc/substitute-in-region)
  (evil-define-key* 'normal 'global
    "x"          #'+vimrc/delete-char-without-yank
    "s"          #'+vimrc/substitute-without-yank
    "c"          #'+vimrc/change-without-yank
    "0"          #'+vimrc/beginning-of-line-or-indentation
    "i"          #'+vimrc/insert
    "A"          #'+vimrc/append-line
    (kbd "C-s")  #'+vimrc/substitute-in-buffer
    "z="         #'+vimrc/close-folds-except-cursor
    "[b"         #'next-buffer
    "]b"         #'previous-buffer)
  (evil-define-key* 'operator 'global
    "A" #'evil-end-of-line)
  (evil-define-key* 'insert 'global
    (kbd "C-f") #'forward-char
    (kbd "C-b") #'backward-char
    (kbd "C-a") #'+vimrc/insert-beginning-of-line-or-indentation
    (kbd "C-e") #'end-of-line)
  (evil-define-key* '(normal visual operator insert) 'global
    [left]  #'ignore
    [right] #'ignore
    [up]    #'ignore
    [down]  #'ignore))
#+end_src

** 全角文字への f

Neovim の =fj= と同じく、 =f= のあとに =j= を打つと、もう1文字読んで digraph で全角文字に変換してから探す。
Emacs では =fj= をキーに割り当てると =f= 単体が使えなくなるため、 =f= 自体を置きかえる。

#+begin_src emacs-lisp :tangle init.el
(setq evil-digraphs-table-user
      '(((?j ?j) . ?j)
        ((?j ?\() . ?（)
        ((?j ?\)) . ?）)
        ((?j ?\[) . ?「)
        ((?j ?\]) . ?」)
        ((?j ?,) . ?、)
        ((?j ?.) . ?。)
        ((?j ?!) . ?！)
        ((?j ??) . ?？)
        ((?j ?~) . ?～)))

(defun +vimrc--read-find-char ()
  (let ((char (evil-read-key)))
    (if (eq char ?j)
        (let ((next (evil-read-key)))
          (require 'evil-digraphs)
          (or (evil-digraph (list ?j next)) next))
      char)))

(with-eval-after-load 'evil
  (evil-define-motion +vimrc/find-char (count char)
    :type inclusive
    (interactive "<c>" (list (+vimrc--read-find-char)))
    (evil-find-char count char))
  (evil-define-key* '(normal visual operator) 'global
    "f" #'+vimrc/find-char))
#+end_src

** エラー一覧の移動

Neovim の =qn= / =qN= は quickfix を移動し、続けて =n= / =N= を押すとさらに移動する。
Emacs では =next-error= で同じことをする。
=q= をプレフィックスにするとマクロの記録ができなくなるため、 =q= のあとのキーが =n= / =N= のときだけ移動し、それ以外はマクロの記録に回す。
=Q= は quickfix を開く代わりに、 =next-error= の対象バッファを表示する。

#+begin_src emacs-lisp :tangle init.el
(defvar-keymap +vimrc-next-error-repeat-map
  "n" #'next-error
  "N" #'previous-error)

(defun +vimrc/display-next-error-buffer ()
  (interactive)
  (display-buffer (or (next-error-find-buffer)
                      (user-error "No buffer to navigate errors in"))))

(with-eval-after-load 'evil
  (evil-define-command +vimrc/record-macro-or-navigate-errors (register)
    :keep-visual t
    :suppress-operator t
    (interactive
     (list (unless (and evil-this-macro defining-kbd-macro)
             (or evil-this-register (evil-read-key)))))
    (pcase register
      (?n (next-error)
          (set-transient-map +vimrc-next-error-repeat-map t))
      (?N (previous-error)
          (set-transient-map +vimrc-next-error-repeat-map t))
      (_ (evil-record-macro register))))
  (evil-define-key* 'normal 'global
    "q" #'+vimrc/record-macro-or-navigate-errors
    "Q" #'+vimrc/display-next-error-buffer))
#+end_src

** タブの移動

Neovim の =gt= / =gT= は、続けて =t= / =T= を押すとさらにタブを移動する。
Emacs のタブは =tab-bar= なので、その移動に同じ繰り返しを付ける。

#+begin_src emacs-lisp :tangle init.el
(defvar-keymap +vimrc-tab-repeat-map
  "t" #'tab-bar-switch-to-next-tab
  "T" #'tab-bar-switch-to-prev-tab)

(defun +vimrc/switch-tab-next ()
  (interactive)
  (tab-bar-switch-to-next-tab)
  (set-transient-map +vimrc-tab-repeat-map t))

(defun +vimrc/switch-tab-previous ()
  (interactive)
  (tab-bar-switch-to-prev-tab)
  (set-transient-map +vimrc-tab-repeat-map t))

(with-eval-after-load 'evil
  (evil-define-key* 'normal 'global
    "gt" #'+vimrc/switch-tab-next
    "gT" #'+vimrc/switch-tab-previous))
#+end_src

** AZIK のかな入力規則カスタマイズ

AZIK ロード直後に呼ばれる =skk-azik-load-hook= で、AZIK 既定規則を上書きする。これは skkeleton の register_kanatable と同義の役割を持つ。

=skk-rom-kana-rule-list= は assoc で先勝ちのため、先頭へ挿入することで既定規則を上書きできる。=:= は AZIK 既定の「ー」を上書きし、変換ポイント開始（=Q= と同等）に割り当てる。

=:= を =skk-set-henkan-point-key= や =skk-downcase-alist= には追加しない。これらは送り検出用であり、追加すると促音判定を経て「っ」が挿入されてしまうためである。

最後に =skk-rule-tree= を nil にし、次回 =skk-mode= 起動時に規則を再コンパイルさせる。

#+begin_src emacs-lisp :tangle init.el
(add-hook 'skk-azik-load-hook
          (lambda ()
            (dolist (rule '((":"   nil skk-set-henkan-point-subr)
                            ("xx"  "x" ("ッ" . "っ"))
                            ("xxa" nil ("ァ" . "ぁ"))
                            ("xxi" nil ("ィ" . "ぃ"))
                            ("xxu" nil ("ゥ" . "ぅ"))
                            ("xxe" nil ("ェ" . "ぇ"))
                            ("xxo" nil ("ォ" . "ぉ"))
                            ("xxya" nil ("ャ" . "ゃ"))
                            ("xxyu" nil ("ュ" . "ゅ"))
                            ("xxyo" nil ("ョ" . "ょ"))
                            ("xxwa" nil ("ヮ" . "ゎ"))
                            ("rr"  nil ("レル" . "れる"))
                            ("nn"  nil ("ナン" . "なん"))))
              (add-to-list 'skk-rom-kana-rule-list rule))
            (setq skk-rule-tree nil)))
#+end_src
